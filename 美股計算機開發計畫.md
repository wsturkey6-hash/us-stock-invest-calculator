# 美股購買計算機 — Claude Code 開發計畫

## 專案概述

建立一個靜態網頁，讓使用者輸入投資總金額與各標的投入比例，自動根據即時股價計算可購買股數。

---

## 技術架構決策

- **形式**：純靜態網頁（單一 HTML + CSS + JS，無需後端）
- **股價 API**：~~Yahoo Finance 非官方 API（免費、無需 API Key）~~ → **已改用 Finnhub**
  - 原因：allorigins.win / corsproxy.io 等免費 CORS Proxy 均已失效（付費或關閉）；Yahoo Finance 本身無 CORS header，無法由瀏覽器直接呼叫
  - 端點：`https://finnhub.io/api/v1/quote?symbol={SYMBOL}&token={API_KEY}`（報價）
  - 端點：`https://finnhub.io/api/v1/stock/profile2?symbol={SYMBOL}&token={API_KEY}`（公司名稱）
  - 免費申請：[finnhub.io/register](https://finnhub.io/register)，免費方案 60 req/min
  - API Key 儲存於 `localStorage`（key: `finnhubKey`）
- **資料持久化**：`localStorage`（儲存追蹤清單）

### Task 2.2 — 股價抓取函式

```javascript
async function fetchStockPrice(symbol) {
  const yahooUrl = `https://query1.finance.yahoo.com/v8/finance/chart/${symbol}`;
  const proxyUrl = `https://api.allorigins.win/get?url=${encodeURIComponent(yahooUrl)}`;
  
  const response = await fetch(proxyUrl);
  const data = await response.json();
  const parsed = JSON.parse(data.contents);
  
  const meta = parsed.chart.result[0].meta;
  return {
    price: meta.regularMarketPrice,
    name: meta.shortName || meta.longName || symbol
  };
}
// 若發生任何錯誤，catch 後回傳 null，UI 顯示「無法取得」
```

### Task 2.3 — 新增標的函式

```javascript
async function addStock() {
  const symbol = document.getElementById("stock-input").value.trim().toUpperCase();
  const errorEl = document.getElementById("add-error");

  // 驗證 1：不為空
  if (!symbol) { errorEl.textContent = "請輸入股票代號"; return; }

  // 驗證 2：不重複
  if (watchlist.find(s => s.symbol === symbol)) {
    errorEl.textContent = `${symbol} 已在追蹤清單中`;
    return;
  }

  errorEl.textContent = "載入中...";

  try {
    const result = await fetchStockPrice(symbol);
    if (!result) throw new Error("查無此代號");

    watchlist.push({ symbol, name: result.name, price: result.price, checked: false, percent: 0 });
    saveToLocalStorage();
    renderTable();
    errorEl.textContent = "";
    document.getElementById("stock-input").value = "";
  } catch (e) {
    errorEl.textContent = `❌ 無法找到股票代號：${symbol}`;
  }
}
```

function validateAllocation() {
  const total = watchlist
    .filter(s => s.checked)
    .reduce((sum, s) => sum + s.percent, 0);

  const warning = document.getElementById("allocation-warning");
  if (total > 100) {
    warning.classList.remove("hidden");
    return false;
  } else {
    warning.classList.add("hidden");
    return true;
  }
}

### Task 2.8 — 核心計算函式

```javascript
function calculate() {
  const budget = parseFloat(document.getElementById("total-budget").value);
  const selected = watchlist.filter(s => s.checked && s.percent > 0 && s.price > 0);

  let summaryHTML = `<table><thead><tr>
    <th>股票代號</th><th>公司名稱</th><th>投入金額</th>
    <th>股價</th><th>可購買股數</th><th>實際花費</th><th>剩餘現金</th>
  </tr></thead><tbody>`;

  selected.forEach(stock => {
    const investAmount = budget * (stock.percent / 100);
    const shares = Math.floor(investAmount / stock.price);
    const actualCost = shares * stock.price;
    const remainder = investAmount - actualCost;

    // 同步更新列表欄位
    stock.shares = shares;
    stock.investAmount = investAmount;

    summaryHTML += `<tr>
      <td>${stock.symbol}</td>
      <td>${stock.name}</td>
      <td>$${investAmount.toFixed(2)}</td>
      <td>$${stock.price.toFixed(2)}</td>
      <td><strong>${shares} 股</strong></td>
      <td>$${actualCost.toFixed(2)}</td>
      <td>$${remainder.toFixed(2)}</td>
    </tr>`;
  });

  summaryHTML += "</tbody></table>";
  document.getElementById("summary-content").innerHTML = summaryHTML;
  document.getElementById("summary-section").classList.remove("hidden");
  renderTable(); // 刷新列表中的「可購買股數」欄
}
```

          placeholder="0" onchange="onPercentChange('${stock.symbol}', this.value)"> %</td>

### Task 2.10 — 刷新股價與自動更新

```javascript
async function fetchAndUpdatePrice(symbol) {
  try {
    const result = await fetchStockPrice(symbol);
    const stock = watchlist.find(s => s.symbol === symbol);
    if (stock && result) {
      stock.price = result.price;
      stock.name = result.name;
      saveToLocalStorage();
      renderTable();
    }
  } catch (e) { /* 靜默失敗，保留舊價格 */ }
}

async function refreshAllPrices() {
  await Promise.all(watchlist.map(s => fetchAndUpdatePrice(s.symbol)));
  document.getElementById("last-updated").textContent =
    `最後更新：${new Date().toLocaleTimeString("zh-TW")}`;
}

// 每 60 秒自動刷新
setInterval(refreshAllPrices, 60000);
```

### Task 2.13 — Finnhub API Key 管理（取代原 Task 2.2 的 Yahoo Finance）

```javascript
const FINNHUB = 'https://finnhub.io/api/v1';

function getApiKey() {
  return localStorage.getItem('finnhubKey') || '';
}

function saveApiKey() {
  const key = document.getElementById('api-key-input').value.trim();
  if (!key) return;
  localStorage.setItem('finnhubKey', key);
  document.getElementById('no-key-banner').classList.add('hidden');
  if (watchlist.length > 0) refreshAllPrices();
}

// 新增標的時：同時取報價（price）與公司名稱（profile2）
async function fetchStockData(symbol) {
  const key = getApiKey();
  if (!key) throw new Error('請先設定 Finnhub API Key');
  const opts = { signal: AbortSignal.timeout(10000) };
  // ⚠️ 原用 profile2，但 ETF 的 profile2 為空，改用 search 端點
  // search.result[].description 對股票與 ETF 均有名稱
  const [quoteResp, searchResp] = await Promise.all([
    fetch(`${FINNHUB}/quote?symbol=${symbol}&token=${key}`, opts),
    fetch(`${FINNHUB}/search?q=${symbol}&token=${key}`, opts),
  ]);
  const quote = await quoteResp.json();
  const search = await searchResp.json().catch(() => ({ result: [] }));
  if (!quote.c || quote.c === 0) throw new Error('查無此股票代號');
  // 精確匹配代號，取 description（如 "VANGUARD TOTAL STOCK MARKET ETF"）
  const match = search.result?.find(r => r.symbol === symbol || r.displaySymbol === symbol);
  const name = match?.description || symbol;
  return { price: quote.c, name };
}

// 刷新股價時：只取報價（省一次 API 呼叫）
async function fetchPriceOnly(symbol) {
  const key = getApiKey();
  if (!key) return null;
  try {
    const resp = await fetch(`${FINNHUB}/quote?symbol=${symbol}&token=${key}`,
      { signal: AbortSignal.timeout(8000) });
    const quote = await resp.json();
    return quote.c || null;
  } catch { return null; }
}
```

### Task 2.14 — 拖曳排序（Drag & Drop Reorder）

在 `app.js` 頂部加入全域變數：

```javascript
let dragSrcIndex = null;
```

拖曳事件處理函式（放在 `renderTable` 之後）：

```javascript
function onDragStart(e, index) {
  dragSrcIndex = index;
  e.dataTransfer.effectAllowed = 'move';
  e.dataTransfer.setData('text/plain', index); // Firefox 需要
  setTimeout(() => {
    document.querySelectorAll('#watchlist-body tr')[index]?.classList.add('dragging');
  }, 0);
}

function onDragOver(e, index) {
  e.preventDefault();
  e.dataTransfer.dropEffect = 'move';
}

function onDragEnter(e, index) {
  if (index === dragSrcIndex) return;
  document.querySelectorAll('#watchlist-body tr').forEach((r, i) => {
    r.classList.toggle('drag-over', i === index);
  });
}

function onDrop(e, index) {
  e.preventDefault();
  if (dragSrcIndex === null || dragSrcIndex === index) return;
  const [moved] = watchlist.splice(dragSrcIndex, 1);
  watchlist.splice(index, 0, moved);
  dragSrcIndex = null;
  saveToLocalStorage();
  renderTable();
}

function onDragEnd() {
  dragSrcIndex = null;
  document.querySelectorAll('#watchlist-body tr').forEach(r => {
    r.classList.remove('dragging', 'drag-over');
  });
}
```

`renderTable` 的 `<tr>` 需加入屬性與拖曳 handle 欄位：

```javascript
// map callback 改為接收 (stock, idx)
tbody.innerHTML = watchlist.map((stock, idx) => `
  <tr class="${stock.checked ? 'selected-row' : ''}"
      draggable="true"
      ondragstart="onDragStart(event, ${idx})"
      ondragover="onDragOver(event, ${idx})"
      ondragenter="onDragEnter(event, ${idx})"
      ondrop="onDrop(event, ${idx})"
      ondragend="onDragEnd()">
    <td class="drag-handle" title="拖曳排序">⠿</td>
    <!-- 其餘欄位不變 -->
  </tr>
`).join('');
```
